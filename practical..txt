package com.mycompany.calculator;

import java.io.IOException;
import java.io.PrintWriter;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

@WebServlet("/CalcServlet")
public class CalcServlet extends HttpServlet {

    @Override
    protected void doPost(HttpServletRequest request,
                          HttpServletResponse response)
                          throws ServletException, IOException {

        response.setContentType("text/html;charset=UTF-8");

        try (PrintWriter out = response.getWriter()) {
            double n1 = Double.parseDouble(request.getParameter("t1"));
            double n2 = Double.parseDouble(request.getParameter("t2"));
            String operator = request.getParameter("btn");
            double result = 0;

            switch (operator) {
                case "+": result = n1 + n2; break;
                case "-": result = n1 - n2; break;
                case "*": result = n1 * n2; break;
                case "/":
                    if (n2 == 0) {
                        out.println("<h2>Cannot divide by zero.</h2>");
                        return;
                    }
                    result = n1 / n2;
                    break;
            }

            out.println("<h2>Result: " + result + "</h2>");
            out.println("<a href='index.html'>Calculate again</a>");
        }
    }
}

import java.io.IOException;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;

import jakarta.servlet.http.HttpServletResponse;

    protected void doPost(HttpServletRequest request,
                          HttpServletResponse response)
                          throws ServletException, IOException {

        response.setContentType("text/html;charset=UTF-8");
        PrintWriter out = response.getWriter();

        String password = request.getParameter("p1");

        if ("Servlet".equals(password)) {
            RequestDispatcher rd =
                    request.getRequestDispatcher("Welcome");

            rd.forward(request, response);

        } else {
            out.println("<h3>Invalid Login</h3>");

            RequestDispatcher rd =
                    request.getRequestDispatcher("index.html");

            rd.include(request, response);
        }
    }

Objective: Create a Servlet that uses a browser cookie named vc to count how many times the user has visited it.
- First visit: create vc = 1
- Later visits: read vc, increase it by 1, and save it again
- Cookie expires after 30 seconds
- Display the current visit count
The PDF contains this code idea, but it assumes the first received cookie is always vc. The version below safely searches for the cookie by name. C:\Users\patel\Downloads\EJ Practical 3a.pdf
In NetBeans, create a Web Application, then create one Servlet:
- Class name: TestCookies
- URL pattern: /TestCookies

import java.io.IOException;
import java.io.PrintWriter;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.Cookie;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

@WebServlet("/TestCookies")
public class TestCookies extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest request,
                         HttpServletResponse response)
                         throws ServletException, IOException {

        response.setContentType("text/html;charset=UTF-8");
        PrintWriter out = response.getWriter();

        int count = 1;
        boolean found = false;

        Cookie[] cookies = request.getCookies();

        if (cookies != null) {
            for (Cookie cookie : cookies) {
                if ("vc".equals(cookie.getName())) {
                    count = Integer.parseInt(cookie.getValue()) + 1;
                    found = true;
                    break;
                }
            }
        }

        Cookie visitCookie = new Cookie("vc", String.valueOf(count));
        visitCookie.setMaxAge(30); // expires after 30 seconds
        response.addCookie(visitCookie);

        out.println("<h2>Visit Count: " + count + "</h2>");

        if (!found) {
            out.println("<p>This is your first visit.</p>");
        }

        out.println("<a href='TestCookies'>Visit again</a>");
    }
}
Run the project and open:
http://localhost:8080/YourProjectName/TestCookies
Refresh the page or click Visit again. It will show Count 1, Count 2, Count 3, etc. After you wait 30 seconds without refreshing, it starts again from Count 1.

package nonblkapp;

import java.io.IOException;
import jakarta.servlet.AsyncContext;
import jakarta.servlet.ReadListener;
import jakarta.servlet.ServletInputStream;

public class ReadingListener implements ReadListener {

    private ServletInputStream input;
    private AsyncContext asyncContext;

    public ReadingListener(ServletInputStream input,
                           AsyncContext asyncContext) {
        this.input = input;
        this.asyncContext = asyncContext;
    }

    @Override
    public void onDataAvailable() throws IOException {
        byte[] buffer = new byte[1024];

        while (input.isReady() && !input.isFinished()) {
            int length = input.read(buffer);

            if (length > 0) {
                System.out.println(
                    new String(buffer, 0, length)
                );
            }
        }
    }

    @Override
    public void onAllDataRead() throws IOException {
        asyncContext.complete();
    }

    @Override
    public void onError(Throwable t) {
        t.printStackTrace();
        asyncContext.complete();
    }
}

                          HttpServletResponse response)

        input.setReadListener(
            new ReadingListener(input, asyncContext)
        );

        response.getWriter().println("<h2>File Reader</h2>");

        while ((line = reader.readLine()) != null) {
            writer.write(line);
            writer.newLine();
            writer.flush();

            response.getWriter().println(line + "<br>");
            response.getWriter().flush();
        }

        writer.close();
        reader.close();

        response.getWriter().println("<h3>Reading Completed</h3>");
    }

package mypack;

import java.io.Serializable;

public class CheckerBean implements Serializable {

    private String name;
    private String age;
    private String hob;
    private String email;
    private String gender;
    private String error;

    public CheckerBean() {
        name = "";
        age = "";
        hob = "";
        email = "";
        gender = "";
        error = "";
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getName() {
        return name;
    }

    public void setAge(String age) {
        this.age = age;
    }

    public String getAge() {
        return age;
    }

    public void setHob(String hob) {
        this.hob = hob;
    }

    public String getHob() {
        return hob;
    }

    public void setEmail(String email) {
        this.email = email;
    }

    public String getEmail() {
        return email;
    }

    public void setGender(String gender) {
        this.gender = gender;
    }

    public String getGender() {
        return gender;
    }

    public String getError() {
        return error;
    }

    public boolean validate() {
        boolean result = true;
        error = "";

        if (name == null || name.trim().equals("")) {
            error += "<br>Enter Name";
            result = false;
        }

        try {
            int ageValue = Integer.parseInt(age);

            if (ageValue < 1 || ageValue > 99) {
                error += "<br>Age must be between 1 and 99";
                result = false;
            }

        } catch (Exception e) {
            error += "<br>Enter valid age";
            result = false;
        }

        if (hob == null || hob.trim().equals("")) {
            error += "<br>Select at least one hobby";
            result = false;
        }

        String emailRegex = "^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+$";

        if (email == null || !email.matches(emailRegex)) {
            error += "<br>Enter valid email";
            result = false;
        }

        if (gender == null || gender.trim().equals("")) {
            error += "<br>Select gender";
            result = false;
        }

        return result;
    }
}
